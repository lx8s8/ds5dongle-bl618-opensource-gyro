cmake_minimum_required(VERSION 3.15)
find_package(bouffalo_sdk REQUIRED HINTS $ENV{BL_SDK_BASE})
sdk_set_main_file(main.c)
target_sources(app PRIVATE boot_store.c usb_manager.c quiet.c)
set(TINYCRYPT_DIR ${BL_SDK_BASE}/components/wireless/bluetooth/blestack/src/common/tinycrypt)
target_sources(app PRIVATE ${TINYCRYPT_DIR}/source/sha256.c ${TINYCRYPT_DIR}/source/utils.c)
sdk_add_include_directories(${TINYCRYPT_DIR}/include)
sdk_add_link_options(-Wl,--wrap=printf,--wrap=puts)
sdk_add_include_directories(.)
sdk_add_include_directories(${BL_SDK_BASE}/components/utils/partition)
sdk_add_compile_definitions(-DBOARD_LCTECH_616=1 -DFORCE_FS_MODE=1)
# Build the pairing bridge only when a generated, pinned helper header is
# explicitly supplied. Ordinary manager packages remain unchanged.
if(DEFINED ENV{DS5_BOND_BUILD_DIR} AND NOT "$ENV{DS5_BOND_BUILD_DIR}" STREQUAL "")
  if(DEFINED ENV{DS5_BOND_SOURCE_ROOT} AND NOT "$ENV{DS5_BOND_SOURCE_ROOT}" STREQUAL "")
    set(BOND_DIR "$ENV{DS5_BOND_SOURCE_ROOT}/experimental/bond_sync")
  else()
    set(BOND_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../experimental/bond_sync")
  endif()
  if(NOT EXISTS "$ENV{DS5_BOND_BUILD_DIR}/bond_extension_build.h")
    message(FATAL_ERROR "Missing generated experimental bond extension header")
  endif()
  sdk_add_compile_definitions(-DDS5_BOND_SYNC_EXPERIMENTAL=1)
  if("$ENV{DS5_BOND_TEST_POWER_WAIT}" STREQUAL "1")
    sdk_add_compile_definitions(-DDS5_BOND_TEST_POWER_WAIT=1)
  endif()
  if("$ENV{DS5_BOND_TEST_FAIL_PROG}" STREQUAL "1")
    sdk_add_compile_definitions(-DDS5_BOND_TEST_FAIL_PROG=1)
  endif()
  sdk_add_include_directories("${BOND_DIR}")
  sdk_add_include_directories("$ENV{DS5_BOND_BUILD_DIR}")
  sdk_add_include_directories("${BL_SDK_BASE}/components/fs/littlefs/littlefs")
  target_sources(app PRIVATE "${BOND_DIR}/extension/assert_guard.c")
  sdk_add_link_options(-Wl,-u,bond_guarded_call,-u,bond_assert_failed)
endif()
# The SDK defaults to a 4 KiB boot stack. Partition validation and the
# pairing synchronizer can exceed that when nested; reserve 16 KiB instead.
# Fail the build if a different SDK changes the expected linker template.
set(BOOT_LINKER_TEMPLATE "${BL_SDK_BASE}/bsp/board/bl616dk/bl616_common.ld.in")
file(READ "${BOOT_LINKER_TEMPLATE}" BOOT_LINKER_CONTENT)
set(BOOT_STACK_OLD ". = . + 0x1000;\n        __StackTop = .;")
set(BOOT_STACK_NEW ". = . + 0x4000;\n        __StackTop = .;")
string(FIND "${BOOT_LINKER_CONTENT}" "${BOOT_STACK_OLD}" BOOT_STACK_OFFSET)
if(BOOT_STACK_OFFSET EQUAL -1)
  message(FATAL_ERROR "Unexpected BL616 linker template: inspect boot stack allocation")
endif()
string(REPLACE "${BOOT_STACK_OLD}" "${BOOT_STACK_NEW}" BOOT_LINKER_CONTENT "${BOOT_LINKER_CONTENT}")
set(BOOT_LINKER_CUSTOM "${CMAKE_CURRENT_BINARY_DIR}/ds5_boot_linker.ld.in")
file(WRITE "${BOOT_LINKER_CUSTOM}" "${BOOT_LINKER_CONTENT}")
sdk_set_linker_script_macro("${BOOT_LINKER_CUSTOM}")
project(ds5_boot_manager)
